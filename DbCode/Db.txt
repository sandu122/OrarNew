--drop table orarNew.entity
create table orarnew.entity (
	id serial primary key
	,name text not null 
	,filt text
	,filt1 text
);

--drop table orarNew.GCluster
create table orarnew.gcluster(
	id serial primary key
	,id_cluster int not null
	,id_group int not null
		,constraint gcluster_fk1 FOREIGN key (id_cluster) references orarnew.entity(id)
		,constraint gcluster_fk2 FOREIGN key (id_group) references orarnew.entity(id)
);

create index gcluster_idx1 on orarnew.gcluster(id_cluster);
create index gcluster_idx2 on orarnew.gcluster(id_group);

--drop table orarnew.std_plan_m
create table orarnew.std_plan_m(
	id serial primary key
	,id_plan text not null
	,plan_name text
);


--drop table orarnew.std_plan_d
create table orarnew.std_plan_d(
	id serial primary key
	,id_plan int not null
	,id_entity int not null
	,curs_cant int not null
	,seminar_cant int not null
	,laborator_cant int not null
		,constraint std_plan_d_fk1 FOREIGN key (id_plan) references orarnew.std_plan_m(id)
		,constraint std_plan_d_fk2 FOREIGN key (id_entity) references orarnew.entity(id)
);

create index std_plan_d_idx1 on orarnew.std_plan_d(id_plan);
create index std_plan_d_idx2 on orarnew.std_plan_d(id_entity);


--drop table orarnew.std_plan_w
create table orarnew.std_plan_w(
	id serial primary key
	,id_plan int not null
	,id_object int not null
	,semestru_p int not null
	,nr_wweeks int not null
		,constraint std_plan_w_fk1 foreign key(id_object) references orarnew.std_plan_d(id)
		,constraint std_plan_w_fk2 foreign key (id_plan) references orarnew.std_plan_m(id)
);

create index std_plan_w_idx1 on orarnew.std_plan_w(id_plan);


--drop table orarnew.gcurricula
create table orarnew.gcurricula (
	id serial primary key
	,id_group int not null
	,id_curricula int not null
		,constraint gcurricula_fk1 FOREIGN KEY (id_group) references orarnew.entity(id)
		,constraint gcurricula_fk2 FOREIGN KEY (id_curricula) references orarnew.std_plan_m(id)
);

create index gcurricula_idx1 on orarnew.gcurricula(id_group);
create index gcurricula_idx2 on orarnew.gcurricula(id_curricula);

create table orarnew.cluster_plan_connection (
	id serial primary key
	,id_lesson_name int not null
	,id_cluster int not null
		,constraint cluster_plan_connection_fk1 foreign key (id_lesson_name) references orarnew.entity(id)
		,constraint cluster_plan_connection_fk2 foreign key (id_cluster) references orarnew.entity(id)
);

create index cluster_plan_connection_idx1 on orarnew.cluster_plan_connection(id_lesson_name);
create index cluster_plan_connection_idx2 on orarnew.cluster_plan_connection(id_cluster);

--Views/functions
select * from orarnew.v2_std_plan_d where id_plan = 1;

 
create or replace view orarnew.v_std_plan_d as
select m.*, e.name
	from orarnew.std_plan_d m
	left join orarnew.entity e on m.id_entity = e.id


select * from orarnew.v2_std_plan_d
create or replace view orarnew.v2_std_plan_d as
select m.*
	,e.name obj_name
	,e.filt1 comun
	,c.name gruop_name
from orarnew.std_plan_d m
left join orarnew.entity e on m.id_entity = e.id

--View pentru unire mai departe	

create or replace view orarnew.v_std_plan_d as
select spd.*
	,e.name
	,e.filt1 comun
from orarnew.std_plan_d spd
left join orarnew.entity e on spd.id_entity = e.id

create or replace view orarnew.gcluster as
select gc.*
	,e.name cluster_name
	,e1.name group_name
from orarnew.gcluster gc
left join orarnew.entity e on gc.id_cluster = e.id
left join orarnew.entity e1 on gc.id_group = e1.id

create or replace view orarnew.cluster_plan_connection as
select cpc.*
	,e.name lesson_name
	,e1.name cluster_name
from orarnew.cluster_plan_connection cpc
left join orarnew.entity e on cpc.id_lesson_name = e.id
left join orarnew.entity e1 on cpc.id_cluster = e1.id
