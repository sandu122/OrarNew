--drop tabel orart.entity
create table orart.entity(
	id serial primary key
	,name text not null
	,filt_1 text
	,filt_2 text
);

create table orart.cluster_group(
	id serial primary key 
	,id_entity int not null
	,id_parent int  
	,relation_type text 
		,constraint cluster_group_fk1 foreign key (id_entity) references orart.entity (id)
		,constraint cluster_group_fk2 foreign key (id_parent) references orart.entity (id)
);

create index cluster_group_idx1 on orart.cluster_group(id_entity);
create index cluster_group_idx2 on orart.cluster_group(id_parent);


create table orart.discipline(
	id serial primary key
	,id_entity int not null 
	,id_entity_t int not null --lesson type
	,cant_p_week int not null
	,date_start date
	,date_stop date
		,constraint discipline_fk1 foreign key (id_entity) references orart.entity(id)
		,constraint discipline_fk2 foreign key (id_entity_t) references orart.entity(id)
);
CREATE UNIQUE INDEX uniq_discipline
ON orart.discipline(id_entity, id_entity_t, cant_p_week);
create index discipline_idx1 on orart.discipline(id_entity);
create index discipline_idx2 on orart.discipline(id_entity_t);

		,constraint prof_lect_cluster_fk1 foreign key (id_entity_p) references orart.entity(id)
		,constraint prof_lect_cluster_fk2 foreign key (id_entity_l) references orart.discipline(id)

create or replace procedure orart.test_conversion()
language plpgsql
as $$
declare
	rec record;
	new_obj_id INT;
	new_teacher_id int;


    grp_list TEXT[];
    grp_name TEXT;
    grp_id INT;
    grp1_id INT;
    grp2_id INT;
    grp3_id INT;

begin
	update orart.temp_table set id_obj = (select id from orart.entity where name = trim(obj_name)),
								id_type = (select id from orart.entity where name = lesson_type),
								id_teacher = (select id from orart.entity where name = teacher_name);
	--
	for rec in 
		select distinct obj_name
		from orart.temp_table where id_obj is null
	loop 
		insert into orart.entity(name, filt_1) values (trim(rec.obj_name), 'lectie') returning id into new_obj_id; 
		UPDATE orart.temp_table
        SET id_obj = new_obj_id
        WHERE obj_name = rec.obj_name; 
	end loop;
	--
	for rec in
		select distinct teacher_name
		from orart.temp_table where id_teacher is null
	loop
		insert into orart.entity(name, filt_1) values (trim(rec.teacher_name), 'prof') returning id into new_teacher_id;
		update orart.temp_table
		set id_teacher = new_teacher_id
		where teacher_name = rec.teacher_name;
	end loop;

    -- Grupe (dinamic până la 3 per rând)
    FOR rec IN
        SELECT obj_name, group_name
        FROM orart.temp_table
        WHERE group_name IS NOT NULL
    LOOP
        grp1_id := NULL;
        grp2_id := NULL;
        grp3_id := NULL;

        -- split doar după virgulă; spațiile se curăță cu trim
        grp_list := string_to_array(rec.group_name, ',');

		-- grupă 1
		IF array_length(grp_list, 1) >= 1 THEN
    		grp_name := trim(grp_list[1]);
    		SELECT id INTO grp_id FROM orart.entity WHERE name = grp_name;

    		IF grp_id IS NULL THEN
        	IF grp_name LIKE '%-%' THEN
            	INSERT INTO orart.entity(name, filt_1)
            	VALUES (grp_name, 'subgrupa')
            	RETURNING id INTO grp_id;
        	ELSE
            	INSERT INTO orart.entity(name, filt_1)
            	VALUES (grp_name, 'grupa')
            	RETURNING id INTO grp_id;
        	END IF;
    	END IF;

    grp1_id := grp_id;
END IF;

        -- grupă 2
        IF array_length(grp_list, 1) >= 2 THEN
            grp_name := trim(grp_list[2]);
            SELECT id INTO grp_id FROM orart.entity WHERE name = grp_name;
            IF grp_id IS NULL THEN
                INSERT INTO orart.entity(name, filt_1)
                VALUES (grp_name, 'grupa')
                RETURNING id INTO grp_id;
            END IF;
            grp2_id := grp_id;
        END IF;

        -- grupă 3
        IF array_length(grp_list, 1) >= 3 THEN
            grp_name := trim(grp_list[3]);
            SELECT id INTO grp_id FROM orart.entity WHERE name = grp_name;
            IF grp_id IS NULL THEN
                INSERT INTO orart.entity(name, filt_1)
                VALUES (grp_name, 'grupa')
                RETURNING id INTO grp_id;
            END IF;
            grp3_id := grp_id;
        END IF;

        -- actualizează rândul curent din temp_table
        UPDATE orart.temp_table
        SET id_group_1 = grp1_id,
            id_group_2 = grp2_id,
            id_group_3 = grp3_id
        WHERE obj_name = rec.obj_name
          AND group_name = rec.group_name;
    END LOOP;
end;
$$

call orart.test_conversion();

CREATE OR REPLACE PROCEDURE orart.load_disciplines()
LANGUAGE plpgsql
AS $$
DECLARE
    rec RECORD;
BEGIN
    FOR rec IN
        SELECT DISTINCT id_obj,
               id_type,
               lesson_count::NUMERIC::INT AS hours
        FROM orart.temp_table
        WHERE id_obj IS NOT NULL
          AND id_type IS NOT NULL
    LOOP
        INSERT INTO orart.discipline(id_entity, id_entity_t, cant_p_week, date_start, date_stop)
        VALUES (rec.id_obj, rec.id_type, rec.hours,
                TO_DATE('01.09.2025','DD.MM.YYYY'),
                TO_DATE('01.09.2027','DD.MM.YYYY'))
        ON CONFLICT DO NOTHING;
    END LOOP;
END;
$$;

call orart.load_disciplines();

CREATE OR REPLACE PROCEDURE orart.load_cluster_group()
LANGUAGE plpgsql
AS $$
DECLARE
    rec RECORD;
    cluster_id INT;
    cluster_name TEXT;
    parent_id INT;
BEGIN
    -- 1. Creăm clustere doar pentru combinații unice de grupe (minim 2 grupe)
    FOR rec IN
        SELECT DISTINCT id_group_1, id_group_2, id_group_3
        FROM orart.temp_table
        WHERE (id_group_2 IS NOT NULL OR id_group_3 IS NOT NULL)
    LOOP
        -- construim denumirea clusterului pe baza combinației de grupe
        cluster_name := 'Cluster_' ||
                        array_to_string(
                            ARRAY[
                                rec.id_group_1::TEXT,
                                rec.id_group_2::TEXT,
                                rec.id_group_3::TEXT
                            ],
                            '_'
                        );

        -- verificăm dacă clusterul există deja
        SELECT id INTO cluster_id
        FROM orart.entity
        WHERE filt_1 = 'cluster'
          AND name = cluster_name;

        -- dacă nu există, îl creăm
        IF cluster_id IS NULL THEN
            INSERT INTO orart.entity(name, filt_1)
            VALUES (cluster_name, 'cluster')
            RETURNING id INTO cluster_id;
        END IF;

        -- legăm grupele la cluster (cl-gr)
        IF rec.id_group_1 IS NOT NULL THEN
            INSERT INTO orart.cluster_group(id_entity, id_parent, relation_type)
            VALUES (rec.id_group_1, cluster_id, 'cl-gr')
            ON CONFLICT DO NOTHING;
        END IF;
        IF rec.id_group_2 IS NOT NULL THEN
            INSERT INTO orart.cluster_group(id_entity, id_parent, relation_type)
            VALUES (rec.id_group_2, cluster_id, 'cl-gr')
            ON CONFLICT DO NOTHING;
        END IF;
        IF rec.id_group_3 IS NOT NULL THEN
            INSERT INTO orart.cluster_group(id_entity, id_parent, relation_type)
            VALUES (rec.id_group_3, cluster_id, 'cl-gr')
            ON CONFLICT DO NOTHING;
        END IF;

        -- actualizăm temp_table cu id_cluster
        UPDATE orart.temp_table
        SET id_cluster = cluster_id
        WHERE id_group_1 IS NOT DISTINCT FROM rec.id_group_1
          AND id_group_2 IS NOT DISTINCT FROM rec.id_group_2
          AND id_group_3 IS NOT DISTINCT FROM rec.id_group_3;
    END LOOP;

    -- 2. Legături subgrupă → grupă (sb-gr)
    FOR rec IN
        SELECT id, name
        FROM orart.entity
        WHERE filt_1 = 'subgrupa'
    LOOP
        -- găsim grupa principală (partea dinainte de '-')
        SELECT id INTO parent_id
        FROM orart.entity
        WHERE name = split_part(rec.name,'-',1)
          AND filt_1 = 'grupa';

        IF parent_id IS NOT NULL THEN
            INSERT INTO orart.cluster_group(id_entity, id_parent, relation_type)
            VALUES (rec.id, parent_id, 'sb-gr')
            ON CONFLICT DO NOTHING;
        END IF;
    END LOOP;
	insert into orart.cluster_group(id_entity, id_parent, relation_type)
select distinct id_parent::INT as id_parent, null::int, 'cl-gr'::text from orart.cluster_group where 
id_parent not in (select id_entity from orart.cluster_group); 
END;
$$;


select * from orart.entity;


call orart.load_cluster_group();

CREATE OR REPLACE PROCEDURE orart.load_prof_lect_cluster()
LANGUAGE plpgsql
AS $$
DECLARE
    rec RECORD;
    disc_id INT;
    hours INT;
    target_c INT;
BEGIN
    -- iterate through distinct rows that can yield lessons
    FOR rec IN
        SELECT DISTINCT
               t.id_teacher,
               t.id_obj,
               t.id_type,
               t.lesson_count,
               t.id_cluster,
               t.id_group_1,
               t.id_group_2,
               t.id_group_3
        FROM orart.temp_table t
        WHERE t.id_teacher IS NOT NULL
          AND t.id_obj     IS NOT NULL
          AND t.id_type    IS NOT NULL
    LOOP
        -- normalize hours
        hours := NULL;
        IF rec.lesson_count IS NOT NULL THEN
            BEGIN
                hours := rec.lesson_count::NUMERIC::INT;
            EXCEPTION WHEN others THEN
                hours := NULL;
            END;
        END IF;

        -- find the discipline id (must exist from load_disciplines)
        SELECT d.id INTO disc_id
        FROM orart.discipline d
        WHERE d.id_entity   = rec.id_obj
          AND d.id_entity_t = rec.id_type
          AND (hours IS NULL OR d.cant_p_week = hours)
        ORDER BY d.id -- deterministic pick if multiple dates
        LIMIT 1;

        -- if discipline missing, skip safely
        IF disc_id IS NULL THEN
            CONTINUE;
        END IF;

        -- case 1: cluster exists -> single record linked to the cluster
        IF rec.id_cluster IS NOT NULL THEN
            target_c := rec.id_cluster;

            INSERT INTO orart.prof_lect_cluster(id_entity_p, id_entity_l, id_entity_c, date_start, date_stop)
            VALUES (rec.id_teacher, disc_id, target_c,
                    DATE '2025-09-01', DATE '2027-09-01')
            ON CONFLICT DO NOTHING;

        ELSE
            -- case 2: no cluster -> create records for each present group/subgroup
            IF rec.id_group_1 IS NOT NULL THEN
                INSERT INTO orart.prof_lect_cluster(id_entity_p, id_entity_l, id_entity_c, date_start, date_stop)
                VALUES (rec.id_teacher, disc_id, rec.id_group_1,
                        DATE '2025-09-01', DATE '2027-09-01')
                ON CONFLICT DO NOTHING;
            END IF;

            IF rec.id_group_2 IS NOT NULL THEN
                INSERT INTO orart.prof_lect_cluster(id_entity_p, id_entity_l, id_entity_c, date_start, date_stop)
                VALUES (rec.id_teacher, disc_id, rec.id_group_2,
                        DATE '2025-09-01', DATE '2027-09-01')
                ON CONFLICT DO NOTHING;
            END IF;

            IF rec.id_group_3 IS NOT NULL THEN
                INSERT INTO orart.prof_lect_cluster(id_entity_p, id_entity_l, id_entity_c, date_start, date_stop)
                VALUES (rec.id_teacher, disc_id, rec.id_group_3,
                        DATE '2025-09-01', DATE '2027-09-01')
                ON CONFLICT DO NOTHING;
            END IF;
        END IF;

    END LOOP;
END;
$$;

call orart.load_prof_lect_cluster();
