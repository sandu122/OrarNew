--drop tabel orart.entity
create table orart.entity(
	id serial primary key
	,name text not null
	,filt_1 text
	,filt_2 text
);

create table orart.cluster_group(
	id serial primary key 
	,id_entity int not null
	,id_parent int  
	,relation_type text 
		,constraint cluster_group_fk1 foreign key (id_entity) references orart.entity (id)
		,constraint cluster_group_fk2 foreign key (id_parent) references orart.entity (id)
);

create index cluster_group_idx1 on orart.cluster_group(id_entity);
create index cluster_group_idx2 on orart.cluster_group(id_parent);


create table orart.discipline(
	id serial primary key
	,id_entity int not null 
	,id_entity_t int not null --lesson type
	,cant_p_week int not null
	,less_lang text
		,constraint discipline_fk1 foreign key (id_entity) references orart.entity(id)
		,constraint discipline_fk2 foreign key (id_entity_t) references orart.entity(id)
);

create index discipline_idx1 on orart.discipline(id_entity);
create index discipline_idx2 on orart.discipline(id_entity_t);

--drop table orart.prof_lect_cluster;
create table orart.prof_lect_cluster(
	id serial primary key 
	,id_entity_p int not null 
	,id_entity_l int not null 
	,id_entity_c int not null 
		,constraint prof_lect_cluster_fk1 foreign key (id_entity_p) references orart.entity(id)
		,constraint prof_lect_cluster_fk2 foreign key (id_entity_l) references orart.discipline(id)
		,constraint prof_lect_cluster_fk3 foreign key (id_entity_c) references orart.entity(id)
);

create index prof_lect_cluster_idx1 on orart.prof_lect_cluster(id_entity_p);
create index prof_lect_cluster_idx2 on orart.prof_lect_cluster(id_entity_l);
create index prof_lect_cluster_idx3 on orart.prof_lect_cluster(id_entity_c);

create or replace view orart.v_discipline as
	select d.*
			,e.name lessonName
			,e1.name lessonType
	from orart.discipline d
		left join orart.entity e on d.id_entity = e.id
		left join orart.entity e1 on d.id_entity_t = e1.id

create or replace view orart.v_cluster_group as
	select cg.*
			,e.name entityName
			,e1.name parentName
	from orart.cluster_group cg
		left join orart.entity e on cg.id_entity = e.id
		left join orart.entity e1 on cg.id_parent = e1.id
		
		
create or replace view orart.v_prof_lect_cluster as
	select plc.*
			,e.name profName
			,d.lessonName
			,d.lessonType
			,d.cant_p_week
			,cg.entityName
			,cg.parentName
	from orart.prof_lect_cluster plc
		left join orart.entity e on plc.id_entity_p = e.id
		left join orart.v_discipline d on plc.id_entity_l = d.id
		left join orart.v_cluster_group cg on plc.id_entity_c = cg.id_entity

create or replace view orart.v_prof_lect_cluster_2	
SELECT plc.id_entity_l   AS LessonId,
       d.lessonName,
       d.lessonType,
       plc.id_entity_p   AS ProfessorId,
       e.name            AS Professor,
       d.cant_p_week,
       cg.entityName     AS TargetName,
       cg.parentName
FROM orart.v_prof_lect_cluster plc
LEFT JOIN orart.entity e ON plc.id_entity_p = e.id
LEFT JOIN orart.v_discipline d ON plc.id_entity_l = d.id
LEFT JOIN orart.v_cluster_group cg ON plc.id_entity_c = cg.id_entity;
